-- 1. LIMPIEZA DE TABLAS PREVIAS
DROP TABLE IF EXISTS notificacion CASCADE;
DROP TABLE IF EXISTS coincidencia CASCADE;
DROP TABLE IF EXISTS reporte CASCADE;
DROP TABLE IF EXISTS objeto CASCADE;
DROP TABLE IF EXISTS categoria CASCADE;
DROP TABLE IF EXISTS estado_coincidencia CASCADE;
DROP TABLE IF EXISTS estado_reporte CASCADE;
DROP TABLE IF EXISTS usuario CASCADE;
DROP TABLE IF EXISTS rol CASCADE;

-- ============================================================
-- 2. TABLAS INDEPENDIENTES (CATÁLOGOS BASE SIN FK)
-- ============================================================

CREATE TABLE rol (
    id_rol SERIAL PRIMARY KEY,
    nombre VARCHAR(30) NOT NULL UNIQUE,
    descripcion VARCHAR(150)
);

CREATE TABLE categoria (
    id_categoria SERIAL PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL UNIQUE,
    descripcion VARCHAR(150),
    activo BOOLEAN NOT NULL DEFAULT TRUE
);

CREATE TABLE estado_reporte (
    id_estado_reporte SERIAL PRIMARY KEY,
    nombre VARCHAR(30) NOT NULL UNIQUE,
    descripcion VARCHAR(150)
);

CREATE TABLE estado_coincidencia (
    id_estado_coincidencia SERIAL PRIMARY KEY,
    nombre VARCHAR(30) NOT NULL UNIQUE,
    descripcion VARCHAR(150)
);

-- ============================================================
-- 3. TABLAS DEPENDIENTES
-- ============================================================

-- Entidad: OBJETO (Clasificado por Categoria)
CREATE TABLE objeto (
    id_objeto SERIAL PRIMARY KEY,
    id_categoria INT NOT NULL REFERENCES categoria(id_categoria),
    nombre VARCHAR(100) NOT NULL,
    descripcion TEXT NOT NULL,
    color VARCHAR(50),
    marca VARCHAR(80),
    activo BOOLEAN NOT NULL DEFAULT TRUE
);

-- Entidad: USUARIO (Asignado a un Rol)
CREATE TABLE usuario (
    id_usuario SERIAL PRIMARY KEY,
    id_rol INT NOT NULL REFERENCES rol(id_rol),
    nombre VARCHAR(50) NOT NULL,
    apellido VARCHAR(50) NOT NULL,
    correo VARCHAR(100) NOT NULL UNIQUE,
    contrasena_hash VARCHAR(255) NOT NULL,
    fecha_registro TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    activo BOOLEAN NOT NULL DEFAULT TRUE
);

-- Entidad: REPORTE (Registrado por Usuario, sobre un Objeto)
CREATE TABLE reporte (
    id_reporte SERIAL PRIMARY KEY,
    id_usuario INT NOT NULL REFERENCES usuario(id_usuario),
    id_objeto INT NOT NULL REFERENCES objeto(id_objeto),
    id_estado_reporte INT NOT NULL REFERENCES estado_reporte(id_estado_reporte),
    tipo_reporte VARCHAR(15) NOT NULL CHECK (tipo_reporte IN ('PERDIDO', 'ENCONTRADO')),
    fecha_evento DATE NOT NULL,
    hora_evento TIME,
    fecha_creacion TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- Entidad: COINCIDENCIA (Empareja dos Reportes)
CREATE TABLE coincidencia (
    id_coincidencia SERIAL PRIMARY KEY,
    id_reporte_perdido INT NOT NULL REFERENCES reporte(id_reporte),
    id_reporte_encontrado INT NOT NULL REFERENCES reporte(id_reporte),
    id_estado_coincidencia INT NOT NULL REFERENCES estado_coincidencia(id_estado_coincidencia),
    porcentaje_similitud DECIMAL(5,2) NOT NULL CHECK (porcentaje_similitud >= 0 AND porcentaje_similitud <= 100),
    detalle_coincidencia TEXT,
    fecha_generacion TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT chk_reportes_diferentes CHECK (id_reporte_perdido <> id_reporte_encontrado),
    CONSTRAINT uq_coincidencia_reportes UNIQUE (id_reporte_perdido, id_reporte_encontrado)
);

-- Entidad: NOTIFICACION (Enviada al Usuario)
CREATE TABLE notificacion (
    id_notificacion SERIAL PRIMARY KEY,
    id_usuario INT NOT NULL REFERENCES usuario(id_usuario) ON DELETE CASCADE,
    id_coincidencia INT REFERENCES coincidencia(id_coincidencia) ON DELETE SET NULL,
    titulo VARCHAR(100) NOT NULL,
    mensaje VARCHAR(255) NOT NULL,
    tipo VARCHAR(30) NOT NULL CHECK (tipo IN ('COINCIDENCIA', 'RECUPERACION', 'SISTEMA')),
    leida BOOLEAN NOT NULL DEFAULT FALSE,
    fecha_creacion TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);
